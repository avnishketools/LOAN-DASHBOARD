  <script>

    const STORAGE_KEY = 'avanish_hub_data_v2';

    const DEFAULT_PITCH = "Hi, here is your verified application link for {TITLE}:\n{URL}\n\nLet me know once you complete the form!";


    let state = JSON.parse(localStorage.getItem(STORAGE_KEY)) || {

      links: [],

      sheets: [],

      notes: "",

      pitchTemplate: DEFAULT_PITCH,

      theme: "light",

      totalCopies: 0,

      masterCompanies: []

    };


    function persist() {

      localStorage.setItem(STORAGE_KEY, JSON.stringify(state));

      updateStats();

    }


    function showToast(msg) {

      const t = document.getElementById('toast');

      t.textContent = msg;

      t.classList.add('show');

      setTimeout(() => t.classList.remove('show'), 2200);

    }


    function applyTheme() {

      document.documentElement.setAttribute('data-theme', state.theme || 'light');

      document.getElementById('themeBtn').textContent = state.theme === 'dark' ? '☀️' : '🌙';

    }

    function toggleTheme() {

      state.theme = state.theme === 'dark' ? 'light' : 'dark';

      applyTheme();

      persist();

    }


    function openModal(id) {

      if (id === 'pitchSettingsModal') {

        document.getElementById('pitchTemplateInput').value = state.pitchTemplate || DEFAULT_PITCH;

      }

      if (id === 'emiModal') calcEMI();

      if (id === 'sheetsModal') renderSheets();

      if (id === 'companySearchPortalModal') renderMasterSearchResults();

      document.getElementById(id).showModal();

    }

    function closeModal(id) { document.getElementById(id).close(); }


    function handleMasterExcelUpload(e) {

      const file = e.target.files[0];

      if (!file) return;

      document.getElementById('uploadStatus').textContent = "Reading multi-bank sheets...";


      const reader = new FileReader();

      reader.onload = function(evt) {

        try {

          const data = new Uint8Array(evt.target.result);

          const workbook = XLSX.read(data, { type: 'array' });

          let allRecords = [];


          workbook.SheetNames.forEach(sheetName => {

            const worksheet = workbook.Sheets[sheetName];

            const json = XLSX.utils.sheet_to_json(worksheet, { defval: "" });


            json.forEach(row => {

              const normRow = {};

              Object.keys(row).forEach(k => {

                normRow[k.trim().toUpperCase()] = row[k];

              });


              const compName = normRow['EMPLOYER_NAME'] || normRow['COMPANY NAME'] || normRow['COMPANY'];

              const category = normRow['EMPLOYER_CATEGORY'] || normRow['CATEGORY'] || normRow['CAT'];


              if (compName) {

                allRecords.push({

                  company: String(compName).trim().toUpperCase(),

                  bank: sheetName.trim(),

                  category: String(category || 'Unclassified').trim()

                });

              }

            });

          });


          state.masterCompanies = allRecords;

          persist();

          document.getElementById('uploadStatus').textContent = `Successfully loaded ${allRecords.length.toLocaleString()} records across ${workbook.SheetNames.length} bank sheets!`;

          renderMasterSearchResults();

          showToast('Master company data loaded!');

        } catch (err) {

          console.error(err);

          document.getElementById('uploadStatus').textContent = "Error parsing excel file.";

          showToast('Failed to read Excel file');

        }

      };

      reader.readAsArrayBuffer(file);

    }


    function renderMasterSearchResults() {

      const q = (document.getElementById('masterCompanySearchInput')?.value || '').toLowerCase().trim();

      const tbody = document.getElementById('masterResultsBody');

      const records = state.masterCompanies || [];


      if (records.length === 0) {

        tbody.innerHTML = `<tr><td colspan="3" style="text-align:center; color:var(--text-muted); padding:24px;">No master company data loaded yet. Please upload your Excel file above.</td></tr>`;

        return;

      }


      const grouped = {};

      records.forEach(r => {

        if (!grouped[r.company]) {

          grouped[r.company] = { name: r.company, cats: new Set(), lenders: [] };

        }

        grouped[r.company].cats.add(r.category);

        grouped[r.company].lenders.push(`${r.bank}: ${r.category}`);

      });


      let list = Object.values(grouped);

      if (q) {

        list = list.filter(item => item.name.toLowerCase().includes(q));

      }


      const displayList = list.slice(0, 100);


      if (displayList.length === 0) {

        tbody.innerHTML = `<tr><td colspan="3" style="text-align:center; color:var(--text-muted); padding:24px;">No matching companies found for "${escapeHTML(q)}"</td></tr>`;

        return;

      }


      tbody.innerHTML = displayList.map(item => `

        <tr>

          <td style="font-weight:600;">${escapeHTML(item.name)}</td>

          <td><span class="badge">${escapeHTML([...item.cats][0] || 'N/A')}</span></td>

          <td style="font-size:12px; color:var(--text-muted);">${escapeHTML(item.lenders.join(' | '))}</td>

        </tr>

      `).join('');

    }


    function openAddLinkModal(category = 'bank') {

      document.getElementById('editId').value = '';

      document.getElementById('linkModalTitle').textContent = 'Add Campaign Link';

      document.getElementById('linkCategory').value = category;

      document.getElementById('linkTitle').value = '';

      document.getElementById('linkUrl').value = '';

      document.getElementById('linkTag').value = '';

      openModal('linkModal');

    }


    function editLink(id) {

      const item = state.links.find(l => l.id === id);

      if (!item) return;

      document.getElementById('editId').value = item.id;

      document.getElementById('linkModalTitle').textContent = 'Edit Link';

      document.getElementById('linkCategory').value = item.category;

      document.getElementById('linkTitle').value = item.title;

      document.getElementById('linkUrl').value = item.url;

      document.getElementById('linkTag').value = item.tag || '';

      openModal('linkModal');

    }


    function saveLink(e) {

      e.preventDefault();

      const id = document.getElementById('editId').value;

      const linkObj = {

        id: id || Date.now().toString(),

        category: document.getElementById('linkCategory').value,

        title: document.getElementById('linkTitle').value.trim(),

        url: document.getElementById('linkUrl').value.trim(),

        tag: document.getElementById('linkTag').value.trim(),

        pinned: id ? (state.links.find(l => l.id === id)?.pinned || false) : false

      };


      if (id) {

        const idx = state.links.findIndex(l => l.id === id);

        if (idx > -1) state.links[idx] = linkObj;

      } else {

        state.links.unshift(linkObj);

      }


      persist();

      renderLinks();

      closeModal('linkModal');

      showToast('Link saved!');

    }


    function deleteLink(id) {

      state.links = state.links.filter(l => l.id !== id);

      persist();

      renderLinks();

      showToast('Link removed');

    }


    function togglePin(id) {

      const item = state.links.find(l => l.id === id);

      if (item) {

        item.pinned = !item.pinned;

        persist();

        renderLinks();

      }

    }


    function copyRaw(id) {

      const item = state.links.find(l => l.id === id);

      if (!item) return;

      navigator.clipboard.writeText(item.url);

      state.totalCopies = (state.totalCopies || 0) + 1;

      persist();

      showToast('Link copied!');

    }


    function copyPitch(id) {

      const item = state.links.find(l => l.id === id);

      if (!item) return;

      const tpl = state.pitchTemplate || DEFAULT_PITCH;

      const msg = tpl.replace(/\{TITLE\}/g, item.title).replace(/\{URL\}/g, item.url);

      navigator.clipboard.writeText(msg);

      state.totalCopies = (state.totalCopies || 0) + 1;

      persist();

      showToast('Link + Pitch copied!');

    }


    function savePitchTemplate() {

      state.pitchTemplate = document.getElementById('pitchTemplateInput').value.trim() || DEFAULT_PITCH;

      persist();

      closeModal('pitchSettingsModal');

      showToast('Pitch template updated!');

    }


    function showQR(id) {

      const item = state.links.find(l => l.id === id);

      if (!item) return;

      document.getElementById('qrTitle').textContent = item.title;

      document.getElementById('qrUrlText').textContent = item.url;

      document.getElementById('qrImage').src = `https://api.qrserver.com/v1/create-qr-code/?size=220x220&data=${encodeURIComponent(item.url)}`;

      openModal('qrModal');

    }

      return `

        <article class="card ${item.pinned ? 'pinned' : ''}">

          <div>

            <div class="card-top">

              <div>

                <span class="badge">${item.tag || (item.category === 'bank' ? 'BANK UTM' : 'CAMPAIGN')}</span>

                <h3>${escapeHTML(item.title)}</h3>

              </div>

                <button class="icon-btn" onclick="editLink('${item.id}')" title="Edit">✎</button>

                <button class="icon-btn" onclick="deleteLink('${item.id}')" title="Delete">🗑</button>

            <a class="btn btn-sm" href="${escapeHTML(item.url)}" target="_blank" rel="noopener">↗</a>

          </div>

        </article>

      `;

    }


    function renderLinks() {

      const q = document.getElementById('searchInput').value.toLowerCase().trim();

      const filtered = state.links.filter(l =>

        l.title.toLowerCase().includes(q) ||

        l.url.toLowerCase().includes(q) ||

        (l.tag && l.tag.toLowerCase().includes(q))

      );


      const pinned = filtered.filter(l => l.pinned);

      const banks = filtered.filter(l => l.category === 'bank');

      const others = filtered.filter(l => l.category === 'other');


      const pinnedSec = document.getElementById('pinnedSection');

      if (pinned.length > 0) {

        pinnedSec.style.display = 'block';

        document.getElementById('pinnedGrid').innerHTML = pinned.map(createCardHTML).join('');

      } else {

        pinnedSec.style.display = 'none';

      }


      document.getElementById('bankGrid').innerHTML = banks.length

        ? banks.map(createCardHTML).join('')

        : `<div class="empty-state">No Bank UTM links yet. Click "+ Add Bank Link" to save one.</div>`;


      document.getElementById('otherGrid').innerHTML = others.length

        ? others.map(createCardHTML).join('')

        : `<div class="empty-state">No Other UTM links yet. Click "+ Add Other Link" to save one.</div>`;


      updateStats();

    }


    function updateStats() {

      document.getElementById('statTotal').textContent = state.links.length;

      document.getElementById('statCompaniesCount').textContent = (state.masterCompanies ? new Set(state.masterCompanies.map(r => r.company)).size : 0).toLocaleString();

      document.getElementById('statBanksCount').textContent = (state.masterCompanies ? new Set(state.masterCompanies.map(r => r.bank)).size : 0);

    }


    function calcEMI() {

      const P = parseFloat(document.getElementById('emiAmount').value) || 0;

      const annualRate = parseFloat(document.getElementById('emiRate').value) || 0;

      const N = parseInt(document.getElementById('emiMonths').value) || 1;

      const R = annualRate / 12 / 100;


      let emi = (R === 0) ? (P / N) : (P * R * Math.pow(1 + R, N)) / (Math.pow(1 + R, N) - 1);

      const total = emi * N;

      const interest = total - P;


      document.getElementById('emiMonthly').textContent = '₹' + Math.round(emi).toLocaleString('en-IN');

      document.getElementById('emiInterest').textContent = '₹' + Math.round(interest).toLocaleString('en-IN');

      document.getElementById('emiTotal').textContent = '₹' + Math.round(total).toLocaleString('en-IN');

    }


    function renderSheets() {

        return;

      }

      box.innerHTML = state.sheets.sheets.map((s, idx) => `...`).join(''); // handled safely

    }


    function escapeHTML(str) {

      return String(str).replace(/[&<>"']/g, m => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]));

    }

        e.preventDefault();

        document.getElementById('searchInput').focus();

      }

    });


    applyTheme();
